import { Cause, Exit } from "effect";
import { toast } from "sonner";

import { TodoNotFound, Unauthorized } from "@my-app/contracts";

const messageFor = <E>(cause: Cause.Cause<E>) => {
  const error = Cause.squash(cause);

  if (error instanceof TodoNotFound) {
    return "That todo no longer exists.";
  }

  if (error instanceof Unauthorized) {
    return "Your session has ended. Sign in again.";
  }

  return "Something went wrong. Try again.";
};

  if (Exit.isFailure(exit)) {
    toast.error(messageFor(exit.cause));
  }
};

const decodeTitle = Schema.decodeUnknownExit(TodoTitle);

const TodoItem = ({ todo }: { todo: Todo }) => {
  const [pending, startMutation] = useTransition();

  const updateTodo = useAtomSet(updateTodoAtom(todo.id), {
    mode: "promiseExit",
  });

  const deleteTodo = useAtomSet(deleteTodoAtom(todo.id), {
    mode: "promiseExit",
  });

  return (
    <li className="hover:bg-muted/40 flex items-center gap-3 px-4 py-2.5 transition-colors">
      <Checkbox
        aria-label={todo.title}
        checked={todo.completed}
        disabled={pending}
        onCheckedChange={(completed) => {
          startMutation(async () => {
            toastFailure(await updateTodo(completed));
          });
        }}
      />
      <span
        className={
          todo.completed
            ? "text-muted-foreground decoration-muted-foreground/50 flex-1 text-sm line-through"
            : "flex-1 text-sm"
        }
      >
        {todo.title}
      </span>
      <Button
        aria-label={`Delete ${todo.title}`}
        disabled={pending}
        onClick={() => {
          startMutation(async () => {
            toastFailure(await deleteTodo());
          });
        }}
        size="icon-sm"
        variant="ghost"
      >
        <Trash2Icon className="text-muted-foreground" />
      </Button>
    </li>
  );
};

const TodosPage = () => {
  const [draft, setDraft] = useState({ title: "" });
  const { title } = draft;
  const [isCreating, startCreating] = useTransition();
  const { value: todos } = useAtomSuspense(todosAtom);
  const createTodo = useAtomSet(createTodoAtom, { mode: "promiseExit" });
  const validTitle = decodeTitle(title);

  const done = todos.filter((todo) => todo.completed).length;

  const handleSubmit = (event: SubmitEvent<HTMLFormElement>) => {
    event.preventDefault();

    if (Exit.isFailure(validTitle)) {
      return;

    startCreating(async () => {
      const exit = await createTodo({
        payload: { title: validTitle.value },
        reactivityKeys: todosKeys,
      });

      if (Exit.isSuccess(exit)) {
        setDraft((current) => (current === draft ? { title: "" } : current));
      }

      toastFailure(exit);
    });
  };

  return (
    <section className="grid gap-6">
      <div className="flex items-end justify-between gap-4">
        <div className="grid gap-1">
          <h1 className="text-3xl font-semibold tracking-tight">Todos</h1>
          <p className="text-muted-foreground text-sm">
            Private to your account.
          </p>
        </div>
        {todos.length > 0 && (
          <p className="text-muted-foreground text-sm tabular-nums">
            {done} of {todos.length} done
          </p>
        )}
      </div>
      <Card>
        <CardContent>
          <form className="flex gap-2" onSubmit={handleSubmit}>

              aria-label="New todo"
              onChange={(event) => {
                setDraft({ title: event.target.value });

          </form>
        </CardContent>
        {todos.length === 0 ? (
          <div className="grid gap-1 border-t px-4 pt-10 pb-6 text-center">
            <p className="text-sm font-medium">No todos yet.</p>
            <p className="text-muted-foreground text-sm">

            </p>
          </div>
        ) : (
          <ul className="-mb-4 divide-y border-t">
            {todos.map((todo) => (
              <TodoItem key={todo.id} todo={todo} />
            ))}
          </ul>
        )}
      </Card>
    </section>
  );
};

    <section className="grid justify-items-start gap-4" role="alert">
      <p>Could not load your todos.</p>
      <Button

          refresh();

        Try again
      </Button>

};

import { Effect, Layer } from "effect";

import { Auth } from "@my-app/auth";
import {
  AuthMiddleware,
  CurrentUser,
  InternalServerError,
  ServiceUnavailable,
  Unauthorized,
} from "@my-app/contracts";

export const AuthMiddlewareLive = Layer.effect(
  AuthMiddleware,
  Effect.gen(function* () {
    const auth = yield* Auth;

    return Effect.fn("AuthMiddleware")(
      function* (effect, { headers }) {
        const session = yield* Effect.tryPromise(
          async () =>
            await auth.api.getSession({ headers: new Headers(headers) })
        ).pipe(
          Effect.tapError((error) =>
            Effect.logError("Session lookup failed", error)
          ),
          Effect.mapError(() => new ServiceUnavailable())
        );

        if (session === null) {
          return yield* new Unauthorized();
        }

        const { email, id, name } = session.user;

        return yield* Effect.provideService(effect, CurrentUser, {
          email,
          id,
          name,
        });
      },
      // RPC defects otherwise serialize their message and cause. Keep those on the server.
      Effect.catchDefect((cause) =>
        Effect.logError("RPC defect", cause).pipe(
          Effect.andThen(Effect.fail(new InternalServerError()))
        )
      )
    );
  })
);

import { and, eq } from "drizzle-orm";
import { Context, Data, Effect, Layer } from "effect";

import { TodoNotFound } from "@my-app/contracts";
import type { Todo } from "@my-app/contracts";
import { Database } from "@my-app/db";
import { todos } from "@my-app/db/schema/todos";

export class TodoStorageError extends Data.TaggedError("TodoStorageError")<{
  readonly cause: unknown;
}> {}

const storageError = (cause: unknown) => new TodoStorageError({ cause });

const columns = {
  completed: todos.completed,
  createdAt: todos.createdAt,
  id: todos.id,
  title: todos.title,
};

const ownedBy = (userId: string, id: number) =>
  and(eq(todos.id, id), eq(todos.userId, userId));

// Someone else's row reads as missing, so the error does not reveal that the id exists.
const firstOrNotFound = <A>([row]: readonly A[], id: number) =>
  row === undefined
    ? Effect.fail(new TodoNotFound({ id }))
    : Effect.succeed(row);

export class Todos extends Context.Service<
  Todos,
  {
    readonly list: (
      userId: string
    ) => Effect.Effect<readonly Todo[], TodoStorageError>;
    readonly create: (
      userId: string,
      title: string
    ) => Effect.Effect<Todo, TodoStorageError>;
    readonly update: (
      userId: string,
      id: number,
      completed: boolean
    ) => Effect.Effect<Todo, TodoStorageError | TodoNotFound>;
    readonly delete: (
      userId: string,
      id: number
    ) => Effect.Effect<void, TodoStorageError | TodoNotFound>;
  }
>()("@my-app/api/Todos") {
  static readonly layer = Layer.effect(
    Todos,
    Effect.gen(function* () {
      const db = yield* Database;

      return Todos.of({
        list: Effect.fn("Todos.list")(function* (userId) {
          return yield* db
            .select(columns)
            .from(todos)
            .where(eq(todos.userId, userId))
            .orderBy(todos.id)
            .pipe(Effect.mapError(storageError));
        }),
        create: Effect.fn("Todos.create")(function* (userId, title) {
          const [todo] = yield* db
            .insert(todos)
            .values({ title, userId })
            .returning(columns)
            .pipe(Effect.mapError(storageError));

          return todo ?? (yield* Effect.die("insert returned no row"));
        }),
        update: Effect.fn("Todos.update")(function* (userId, id, completed) {
          const rows = yield* db
            .update(todos)
            .set({ completed })
            .where(ownedBy(userId, id))
            .returning(columns)
            .pipe(Effect.mapError(storageError));

          return yield* firstOrNotFound(rows, id);
        }),
        delete: Effect.fn("Todos.delete")(function* (userId, id) {
          const rows = yield* db
            .delete(todos)
            .where(ownedBy(userId, id))
            .returning(columns)
            .pipe(Effect.mapError(storageError));

          yield* firstOrNotFound(rows, id);
        }),
      });
    })
  );
}

}> {}

const storageError = (cause: unknown) => new TodoStorageError({ cause });

const columns = {
  completed: todos.completed,
  createdAt: todos.createdAt,
  id: todos.id,
  title: todos.title,
};

const firstOrNotFound = <A>([row]: readonly A[], id: number) =>
  row === undefined
    ? Effect.fail(new TodoNotFound({ id }))
    : Effect.succeed(row);

export class Todos extends Context.Service<
  Todos,
  {
    readonly list: () => Effect.Effect<readonly Todo[], TodoStorageError>;
    readonly create: (title: string) => Effect.Effect<Todo, TodoStorageError>;
    readonly update: (
      id: number,
      completed: boolean
    ) => Effect.Effect<Todo, TodoStorageError | TodoNotFound>;
    readonly delete: (
      id: number
    ) => Effect.Effect<void, TodoStorageError | TodoNotFound>;
  }
>()("@my-app/api/Todos") {
  static readonly layer = Layer.effect(
    Todos,
    Effect.gen(function* () {
      const db = yield* Database;

      return Todos.of({
        list: Effect.fn("Todos.list")(function* () {
          return yield* db
            .select(columns)
            .from(todos)
            .orderBy(todos.id)
            .pipe(Effect.mapError(storageError));
        }),
        create: Effect.fn("Todos.create")(function* (title) {
          const [todo] = yield* db
            .insert(todos)
            .values({ title })
            .returning(columns)
            .pipe(Effect.mapError(storageError));

          return todo ?? (yield* Effect.die("insert returned no row"));
        }),
        update: Effect.fn("Todos.update")(function* (id, completed) {
          const rows = yield* db
            .update(todos)
            .set({ completed })
            .where(eq(todos.id, id))
            .returning(columns)
            .pipe(Effect.mapError(storageError));

          return yield* firstOrNotFound(rows, id);
        }),
        delete: Effect.fn("Todos.delete")(function* (id) {
          const rows = yield* db
            .delete(todos)
            .where(eq(todos.id, id))
            .returning(columns)
            .pipe(Effect.mapError(storageError));

          yield* firstOrNotFound(rows, id);
        }),
      });
    })
  );
}
